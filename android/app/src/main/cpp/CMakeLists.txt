cmake_minimum_required(VERSION 3.22)
project(uvctool)

set(ROOT_DIR ${CMAKE_SOURCE_DIR}/../../../../../)
set(LIBUSB_SRC ${ROOT_DIR}/third_party/libusb)
set(LIBUVC_SRC ${ROOT_DIR}/third_party/libuvc)

# libusb — static, Android-only sources (no udev, no hotplug netlink discovery)
add_library(usb-1.0 STATIC
    ${LIBUSB_SRC}/libusb/core.c
    ${LIBUSB_SRC}/libusb/descriptor.c
    ${LIBUSB_SRC}/libusb/hotplug.c
    ${LIBUSB_SRC}/libusb/io.c
    ${LIBUSB_SRC}/libusb/strerror.c
    ${LIBUSB_SRC}/libusb/sync.c
    ${LIBUSB_SRC}/libusb/os/linux_usbfs.c
    ${LIBUSB_SRC}/libusb/os/events_posix.c
    ${LIBUSB_SRC}/libusb/os/threads_posix.c
    ${LIBUSB_SRC}/libusb/os/linux_netlink.c
)
target_include_directories(usb-1.0 PUBLIC
    ${LIBUSB_SRC}/libusb
    ${LIBUSB_SRC}/libusb/os
    ${LIBUSB_SRC}/android  # config.h for Android
)
# version_describe.h (generated by libusb's own build) lives in config/
target_include_directories(usb-1.0 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/config)
target_compile_definitions(usb-1.0 PRIVATE _GNU_SOURCE)

# libuvc — static, no JPEG
add_library(uvc STATIC
    ${LIBUVC_SRC}/src/ctrl.c
    ${LIBUVC_SRC}/src/ctrl-gen.c
    ${LIBUVC_SRC}/src/device.c
    ${LIBUVC_SRC}/src/diag.c
    ${LIBUVC_SRC}/src/frame.c
    ${LIBUVC_SRC}/src/init.c
    ${LIBUVC_SRC}/src/stream.c
    ${LIBUVC_SRC}/src/misc.c
)
# libuvc_config.h is normally generated by libuvc's own CMake; we keep a checked-in
# copy (v0.0.8, no JPEG) so the submodule stays pristine.
target_include_directories(uvc PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/config
    ${LIBUVC_SRC}/include
    ${LIBUVC_SRC}/src
)
target_link_libraries(uvc usb-1.0)
target_compile_definitions(uvc PRIVATE LIBUVC_HAS_JPEG=0)
# Small iso URBs. usbfs allocates each URB buffer as ONE physically contiguous
# kmalloc; the libuvc default (32 packets x 3 KiB = 96 KiB, order-5) fails with
# ENOMEM ("submiturb failed, errno=12") on a fragmented phone. 8 packets = 24 KiB.
# 128 transfers x 8 x 125 us = 128 ms of buffering, ~3 MB total (48 ms was too
# little: the event thread stalls longer than that and iso packets are lost).
# PUBLIC: the JNI includes libuvc_internal.h, struct layout must match.
target_compile_definitions(uvc PUBLIC
    LIBUVC_NUM_TRANSFER_BUFS=128
    LIBUVC_PACKETS_PER_TRANSFER_MAX=8)

# JNI shared library
add_library(uvctool SHARED uvctool_jni.cpp)
target_link_libraries(uvctool uvc usb-1.0 android log)
# 16 KB page alignment (Android 15+ devices with 16 KB pages; NDK r26 defaults to 4 KB).
target_link_options(uvctool PRIVATE
    "-Wl,-z,max-page-size=16384"
    "-Wl,-z,common-page-size=16384")
